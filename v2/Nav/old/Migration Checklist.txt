python

def parse_nav_file(file, replace: bool) -> tuple[bool, str, dict]:
    """
    Parses AMFI NAV text file uploaded via Streamlit UI.
    Delegates to nav_engine.ingest_nav_bytes_to_db() — no temp-file dance.
    """
    try:
        result = ingest_nav_bytes_to_db(file.getvalue())
        if result.get('ok'):
            msg = f"Imported {result['inserted']} NAV records | Skipped: {result['skipped']}"
            if result['inserted']:
                bump()
            return True, msg, {
                "rows": result['inserted'],
                "skipped": result['skipped'],
                "fund_houses": len(result.get('fund_houses', set())),
            }
        else:
            sflog.record_failure(
                source="manual_upload", stage="parse",
                rta="AMFI", report="NAV", file=getattr(file, "name", ""),
                msg=result.get('reason', 'Unknown error'),
            )
            return False, result.get('reason', 'Unknown error'), {}
    except Exception as e:
        sflog.record_failure(
            source="manual_upload", stage="parse",
            rta="AMFI", report="NAV", file=getattr(file, "name", ""),
            msg=f"File read error: {e}",
        )
        return False, f"File read error: {e}", {}
